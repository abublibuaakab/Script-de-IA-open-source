local function findModels(keys, maxN)
	local found = {}
	for _, d in ipairs(workspace:GetDescendants()) do
		if d:IsA("Model") then
			local n = string.lower(d.Name)
			for _, k in ipairs(keys) do
				if string.find(n, k, 1, true) then
					table.insert(found, d)
					break
				end
			end
			if #found >= (maxN or 5) then break end
		end
	end
	return found
end

local function modelPos(m)
	local ok, cf = pcall(function() return m:GetPivot() end)
	if ok then return cf.Position end
	return nil
end

local function spinModel(m)
	local root = m.PrimaryPart or m:FindFirstChildWhichIsA("BasePart", true)
	if not root then return end
	local av = root:FindFirstChild("BHSpin")
	if not av then
		av = Instance.new("BodyAngularVelocity")
		av.Name = "BHSpin"
		av.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		av.AngularVelocity = Vector3.new(9e9, 0, 9e9)
		av.Parent = root
	end
end

local function myVehicle()
	local h = hum()
	local seat = h and h.SeatPart
	return seat and seat:FindFirstAncestorOfClass("Model") or nil
end

local function char()
	local c = LocalPlayer.Character
	if c and c.Parent then return c end
	return nil
end

local function hrp()
	local c = char()
	return c and c:FindFirstChild("HumanoidRootPart") or nil
end

local function hum()
	local c = char()
	return c and c:FindFirstChildOfClass("Humanoid") or nil
end

local function notify(t, c)
	Rayfield:Notify({ Title = t, Content = c, Duration = 4 })
end

local function playerByName(n)
	n = string.lower(n)
	for _, plr in ipairs(Players:GetPlayers()) do
		if string.find(string.lower(plr.Name), n, 1, true) or string.find(string.lower(plr.DisplayName), n, 1, true) then
			return plr
		end
	end
	return nil
end

local function tryMusic(id)
	local M = game:GetService("ReplicatedStorage").Remotes.ClientMusic
	local n = tonumber(id)
	if not n then return "ID invalido" end
	pcall(function() M:FireServer(n) end)
	task.wait(0.5)
	pcall(function() M:FireServer("Play", n) end)
	task.wait(0.5)
	pcall(function() M:FireServer("rbxassetid://" .. n) end)
	task.wait(0.5)
	local h = hum()
	local seat = h and h.SeatPart
	if seat then
		pcall(function()
			game:GetService("ReplicatedStorage").Remotes.ToggleVehicleMusic:FireServer(n)
		end)
	end
	return "tentativas enviadas"
end

				if #balls == 0 then return end
				local b = balls[1]
				spinModel(b)
				local me = hrp()
				local bp = modelPos(b)
				if me and bp and (bp - me.Position).Magnitude > 8 then
					b:PivotTo(me.CFrame * CFrame.new(0, 0, -3))
					task.wait(0.5)
				end
				local best, bestD = nil, 60
				for _, plr in ipairs(Players:GetPlayers()) do
					if plr ~= LocalPlayer and plr.Character then
						local ee = plr.Character:FindFirstChild("HumanoidRootPart")
						local mp = modelPos(b)
						if ee and mp then
							local d = (ee.Position - mp).Magnitude
							if d < bestD then best, bestD = ee, d end
						end
					end
				end
				if best then
					b:PivotTo(best.CFrame * CFrame.new(0, 5, 0))
				end
			end)
		end
	end
end)

local TabE = Window:CreateTab("ESP", 0)
local ESPOn = false
local boxes = {}

TabE:CreateToggle({
	Name = "ESP jogadores",
	CurrentValue = false,
	Callback = function(v) ESPOn = v end,
})

TabE:CreateToggle({
	Name = "WALLHACK (contorno)",
	CurrentValue = false,
	Callback = function(v) S.Wallhack = v end,
})

local hls = {}
task.spawn(function()
	while true do
		task.wait(2)
		for _, h in pairs(hls) do pcall(function() h:Destroy() end) end
		hls = {}
		if S.Wallhack then
			pcall(function()
				for _, plr in ipairs(Players:GetPlayers()) do
					if plr ~= LocalPlayer and plr.Character then
						local hl = Instance.new("Highlight")
						hl.FillColor = Color3.fromRGB(255, 80, 80)
						hl.OutlineColor = Color3.fromRGB(255, 255, 255)
						hl.FillTransparency = 0.5
						hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
						hl.Adornee = plr.Character
						hl.Parent = plr.Character
						table.insert(hls, hl)
					end
				end
			end)
		end
	end
end)

RunService.RenderStepped:Connect(function()
	local cam = workspace.CurrentCamera
	if not cam then return end
	for _, b in pairs(boxes) do b:Remove() end
	boxes = {}
	if ESPOn then
		local mePos = nil
		pcall(function() mePos = hrp().Position end)
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and plr.Character then
				local e = plr.Character:FindFirstChild("HumanoidRootPart")
				if e then
					local sp, on = cam:WorldToViewportPoint(e.Position)
					if on then
						local t = Drawing.new("Text")
						t.Size = 14
						t.Center = true
						t.Outline = true
						t.Color = Color3.fromRGB(255, 255, 255)
						t.Position = Vector2.new(sp.X, sp.Y)
						local d = mePos and (" " .. math.floor((e.Position - mePos).Magnitude) .. "m") or ""
						t.Text = plr.DisplayName .. d
						t.Visible = true
						table.insert(boxes, t)
					end
				end
			end
		end
	end
end)
